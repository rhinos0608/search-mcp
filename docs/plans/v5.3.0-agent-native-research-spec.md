### `src/research/strategies/registry.ts`

```typescript
import type { ResearchStrategy, StrategyContext, StrategyFactory } from './types.js';

export interface StrategyInfo {
  name: string;
  description: string;
  requiresLlm: boolean;
}

export class StrategyRegistry {
  private factories = new Map<string, StrategyFactory>();

  register(name: string, factory: StrategyFactory): void {
    if (this.factories.has(name)) {
      throw new Error(`Strategy already registered: ${name}`);
    }
    this.factories.set(name, factory);
  }

  create(name: string, ctx: StrategyContext): ResearchStrategy {
    const factory = this.factories.get(name);
    if (!factory) throw new Error(`Unknown strategy: ${name}. Available: ${[...this.factories.keys()].join(', ')}`);
    return factory(ctx);
  }

  /** Returns the default strategy name based on context. */
  selectDefault(ctx: StrategyContext): string {
    if (ctx.llm) return 'agent';
    return 'pipeline';
  }

  listAvailable(ctx: StrategyContext): StrategyInfo[] {
    return [...this.factories.entries()].map(([name, factory]) => {
      const s = factory(ctx);
      return { name: s.name, description: s.description, requiresLlm: s.requiresLlm };
    });
  }

  has(name: string): boolean {
    return this.factories.has(name);
  }
}

export const strategyRegistry = new StrategyRegistry();
```

## Phase 2: Agent Strategy

### `src/research/strategies/agentStrategy.ts`

```typescript
export class AgentStrategy implements ResearchStrategy {
  readonly name = 'agent';
  readonly description = 'LLM-driven ReAct agent with tool-calling. Adapts research approach based on findings.';
  readonly requiresLlm = true;

  private maxIterations: number;
  private maxSubIterations: number;
  private tools: AgentTool[];
  private collector: CitationCollector;
  private history: AgentHistoryEntry[] = [];
  private abortSignal?: AbortSignal;

  constructor(ctx: StrategyContext) {
    this.maxIterations = ctx.config.agentMaxIterations ?? 30;
    this.tools = buildAgentTools(ctx);
    this.collector = new CitationCollector();
    this.abortSignal = ctx.abortSignal;
  }

  async analyze(query: string, ctx: StrategyContext): Promise<ResearchResult> {
    const systemPrompt = this.buildSystemPrompt(this.tools);
    let iteration = 0;
    let finalAnswer: string | null = null;

    while (iteration < this.maxIterations) {
      this.checkAborted();
      iteration++;

      const response = await ctx.llm!.chat({
        model: ctx.config.model,
        messages: [
          { role: 'system', content: systemPrompt },
          ...this.history.map(this.formatHistoryEntry),
          { role: 'user', content: this.buildTurnPrompt(query) },
        ],
      });

      const parsed = parseAgentResponse(response.content);

      if (parsed.type === 'answer') {
        finalAnswer = parsed.content;
        break;
      }

      if (parsed.type === 'action') {
        const result = await this.executeTool(parsed.tool, parsed.args, ctx);
        this.history.push({
          role: 'assistant',
          thought: parsed.thought,
          action: parsed.tool,
          args: parsed.args,
        });
        this.history.push({
          role: 'tool',
          tool: parsed.tool,
          content: result.content,
        });
      }

      if (parsed.type === 'error') {
        this.history.push({
          role: 'system',
          error: parsed.message,
        });
      }
    }

    if (!finalAnswer) {
      finalAnswer = await this.synthesizeFallback(ctx);
    }

    return this.buildResult(finalAnswer, query, iteration, ctx);
  }
}
```

### Response Parser

```typescript
interface ParsedResponse {
  type: 'action' | 'answer' | 'error';
  // For 'action':
  thought?: string;
  tool?: string;
  args?: Record<string, unknown>;
  // For 'answer':
  content?: string;
  // For 'error':
  message?: string;
}

function parseAgentResponse(text: string): ParsedResponse {
  const thoughtMatch = text.match(/THOUGHT:\s*([\s\S]*?)(?=\n(?:ACTION|ANSWER|$))/i);
  const actionMatch = text.match(/ACTION:\s*(\S+)/i);
  const argsMatch = text.match(/ARGUMENTS:\s*(\{[\s\S]*?\})/i);
  const answerMatch = text.match(/ANSWER:\s*([\s\S]*)/i);

  if (answerMatch) {
    return {
      type: 'answer',
      content: answerMatch[1].trim(),
      thought: thoughtMatch?.[1]?.trim() ?? '',
    };
  }

  if (actionMatch && argsMatch) {
    let args: Record<string, unknown>;
    try { args = JSON.parse(argsMatch[1]); }
    catch { return { type: 'error', message: 'Invalid JSON in ARGUMENTS' }; }
    return {
      type: 'action',
      thought: thoughtMatch?.[1]?.trim() ?? '',
      tool: actionMatch[1].trim(),
      args,
    };
  }

  if (actionMatch && !argsMatch) {
    return { type: 'error', message: 'ACTION specified but no ARGUMENTS found' };
  }

  return { type: 'error', message: 'Could not parse response. Use THOUGHT/ACTION/ARGUMENTS or ANSWER format.' };
}
```

### `src/research/citationCollector.ts`

```typescript
interface CitationEntry {
  index: number;          // 1-based, globally unique
  url: string;
  title: string;
  snippet: string;
  sourceType: string;
}

export class CitationCollector {
  private entries: CitationEntry[] = [];
  private urlMap = new Map<string, number>(); // url → index
  private nextIndex = 1;

  /** Returns the starting citation index for this batch. Thread-safe. */
  addResults(results: SearchResult[], sourceType: string): number {
    const startIdx = this.nextIndex;
    for (const r of results) {
      const url = r.link ?? r.url ?? '';
      // Dedup: reuse existing index if URL already tracked
      if (url && this.urlMap.has(url)) continue;
      
      const idx = this.nextIndex++;
      if (url) this.urlMap.set(url, idx);
      
      this.entries.push({
        index: idx,
        url,
        title: r.title ?? 'Untitled',
        snippet: r.snippet ?? r.body ?? '',
        sourceType,
      });
    }
    return startIdx;
  }

  findCitation(url: string): number | undefined {
    return this.urlMap.get(url);
  }

  getAll(): CitationEntry[] {
    return [...this.entries];
  }

  getCount(): number {
    return this.entries.length;
  }

  reset(): void {
    this.entries = [];
    // urlMap intentionally preserved for cross-run dedup
  }
}
```

### Registration Pattern

Two patterns exist in `src/server.ts::createServer()`:
- **Pattern A — `server.registerTool()`**: Used by `web_search`, `deep_research`, `health_check`. Direct registration with inline schema.
- **Pattern B — `registerFamily()`**: Used by `browser`, `youtube`, `reddit`, `github`, `packages`, `research`. Each family exports a `registerXxxTool(server, cfg)` function with typed actions.

For PubMed and Wikipedia (standalone, single-action tools), use **Pattern A**. For the research family expansion (arXiv, Semantic Scholar already live there), extend the existing `research` family.

### Agent Tool Execution

The agent strategy's `executeTool()` calls existing tool implementations directly — NOT through MCP protocol. This avoids serialization overhead and gives direct access to typed return values:

```typescript
// In agentTools.ts — executeTool dispatches to existing implementation functions
async function executeTool(
  name: string,
  args: Record<string, unknown>,
  ctx: StrategyContext,
  collector: CitationCollector
): Promise<ToolResult> {
  switch (name) {
    case 'search_web':
      return executeWebSearch(args, ctx, collector);    // calls webSearch() from src/tools/webSearch.ts
    case 'search_arxiv':
      return executeArxivSearch(args, collector);       // calls arxivSearch() from src/tools/arxivSearch.ts
    case 'search_pubmed':
      return executePubmedSearch(args, collector);      // calls searchPubMed() from new src/tools/research/pubmedSearch.ts
    case 'search_wikipedia':
      return executeWikipediaSearch(args, collector);    // calls searchWikipedia() from new src/tools/research/wikipediaSearch.ts
    case 'fetch_page':
      return executeFetchPage(args, collector);         // calls webCrawl()/webRead() with full mode
    case 'fetch_focus':
      return executeFetchFocus(args, ctx, collector);   // calls webCrawl() then LLM extraction
    case 'research_subtopic':
      return executeResearchSubtopic(args, ctx, collector); // spawns parallel sub-AgentStrategies
    default:
      return { content: `Unknown tool: ${name}. Available: ${listTools()}` };
  }
}
```

Each execute function: (1) calls the existing implementation (same function used by MCP tool handlers), (2) registers results in the `CitationCollector` for global citation indexing, (3) formats output as `[N] Title\nURL\n\nSnippet` for LLM consumption. This means the agent uses the **exact same search implementations** as our MCP tools — no duplication.

### PubMed Search

```typescript
// src/tools/research/pubmedSearch.ts

const PUBMED_BASE = 'https://eutils.ncbi.nlm.nih.gov/entrez/eutils';
const PUBMED_EMAIL = process.env.PUBMED_EMAIL ?? 'anonymous@example.com';
const PUBMED_API_KEY = process.env.PUBMED_API_KEY; // optional

async function searchPubMed(query: string, limit = 10): Promise<SearchResult[]> {
  // 1. ESearch: get PMIDs
  const esearchUrl = `${PUBMED_BASE}/esearch.fcgi?db=pubmed&term=${encodeURIComponent(query)}&retmax=${limit}&retmode=json&email=${PUBMED_EMAIL}`;
  const ids = await fetch(esearchUrl).then(r => r.json()).then(j => j.esearchresult.idlist);

  // 2. EFetch: get abstracts
  const efetchUrl = `${PUBMED_BASE}/efetch.fcgi?db=pubmed&id=${ids.join(',')}&retmode=xml&email=${PUBMED_EMAIL}`;
  const xml = await fetch(efetchUrl).then(r => r.text());

  // 3. Parse abstracts from XML
  return parsePubMedXml(xml);
}
```

```typescript
const WIKIPEDIA_API = 'https://en.wikipedia.org/api/rest_v1/page/summary';

async function searchWikipedia(query: string, language = 'en'): Promise<SearchResult[]> {
  // Direct page lookup by title
  const url = `https://${language}.wikipedia.org/api/rest_v1/page/summary/${encodeURIComponent(query)}`;
  const resp = await fetch(url);
  if (!resp.ok) {
    // Fall back to search
    return searchWikipediaFallback(query, language);
  }
  const page = await resp.json();
  return [{
    title: page.title,
    link: page.content_urls?.desktop?.page ?? `https://${language}.wikipedia.org/wiki/${encodeURIComponent(query)}`,
    snippet: page.extract?.slice(0, 500) ?? '',
  }];
}
```
